\bigskip\noindent{\Large\bfseries\color{apren} Test 1. Forma binòmica, conjugat i pla complex}\par\medskip
\iniciTest{complexosT1}{b,c,b,b,d,b,a,b,c,c}
\begin{enumerate}
\pregunta Quina propietat defineix la unitat imaginària $i$?
\begin{enumerate}
\opcio{a}{$i=1$}
\opcio{b}{$i^2=-1$}
\opcio{c}{$i^2=1$}
\opcio{d}{$i=-1$}
\end{enumerate}\feedback
\pregunta Quina és la forma binòmica general d’un nombre complex?
\begin{enumerate}
\opcio{a}{$a+b$}
\opcio{b}{$a+bi$, amb $a,b\in\mathbb{R}$}
\opcio{c}{$ab+i$}
\opcio{d}{$a/b+i$}
\end{enumerate}\feedback
\pregunta Si $z=5-2i$, quina és la part imaginària de $z$?
\begin{enumerate}
\opcio{a}{$5$}
\opcio{b}{$-2i$}
\opcio{c}{$-2$}
\opcio{d}{$2$}
\end{enumerate}\feedback
\pregunta Quin és el conjugat de $z=-3+4i$?
\begin{enumerate}
\opcio{a}{$3-4i$}
\opcio{b}{$-3-4i$}
\opcio{c}{$3+4i$}
\opcio{d}{$4-3i$}
\end{enumerate}\feedback
\pregunta Quant val $z\overline z$ si $z=a+bi$?
\begin{enumerate}
\opcio{a}{$a+b$}
\opcio{b}{$a^2-b^2$}
\opcio{c}{$2ab$}
\opcio{d}{$a^2+b^2$}
\end{enumerate}\feedback
\pregunta Quin dels nombres següents és un imaginari pur?
\begin{enumerate}
\opcio{a}{$3+2i$}
\opcio{b}{$-7i$}
\opcio{c}{$4$}
\opcio{d}{$7$}
\end{enumerate}\feedback
\pregunta Quina operació permet dividir per $c+di\ne0$ en forma binòmica?
\begin{enumerate}
\opcio{a}{Multiplicar numerador i denominador per $c-di$}
\opcio{b}{Multiplicar només el numerador per $c+di$}
\opcio{c}{Canviar $i$ per $-1$}
\opcio{d}{Sumar el conjugat al denominador}
\end{enumerate}\feedback
\pregunta Si $z=3+4i$, quin és $|z|$?
\begin{enumerate}
\opcio{a}{$7$}
\opcio{b}{$5$}
\opcio{c}{$25$}
\opcio{d}{$1$}
\end{enumerate}\feedback
\pregunta Els afixos de $z$ i $\overline z$ són simètrics respecte de:
\begin{enumerate}
\opcio{a}{l’eix imaginari}
\opcio{b}{l’origen}
\opcio{c}{l’eix real}
\opcio{d}{la recta $y=x$}
\end{enumerate}\feedback
\pregunta La igualtat de dos complexos $a+bi=c+di$ equival a:
\begin{enumerate}
\opcio{a}{$a=d$ i $b=c$}
\opcio{b}{$a+b=c+d$}
\opcio{c}{$a=c$ i $b=d$}
\opcio{d}{$ab=cd$}
\end{enumerate}\feedback
\end{enumerate}

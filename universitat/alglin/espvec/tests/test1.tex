\bigskip\noindent{\Large\bfseries\color{apren} Test 1. Espais i subespais}\par\medskip
\iniciTest{espvecT1}{b,c,a,d,b,c,a,b,d,c}
\begin{enumerate}
\pregunta Quin ingredient és imprescindible en la definició d’un espai vectorial?
\begin{enumerate}
\opcio{a}{Una distància entre vectors}
\opcio{b}{Un conjunt, un cos d’escalars i dues operacions compatibles}
\opcio{c}{Una base ordenada fixada}
\opcio{d}{Un producte escalar}
\end{enumerate}\feedback
\pregunta Quin és el vector neutre de la suma en un espai vectorial?
\begin{enumerate}
\opcio{a}{Qualsevol vector de norma 1}
\opcio{b}{El vector més petit}
\opcio{c}{El vector zero}
\opcio{d}{El vector unitari}
\end{enumerate}\feedback
\pregunta Si $W$ és un subespai de $V$, quina condició ha de complir sempre?
\begin{enumerate}
\opcio{a}{$0\in W$ i $W$ és tancat per suma i producte per escalars}
\opcio{b}{$W$ ha de ser finit}
\opcio{c}{$W$ ha de tenir exactament una base}
\opcio{d}{$W$ no pot contenir el vector zero}
\end{enumerate}\feedback
\pregunta Quin conjunt és un subespai de $\mathbb R^2$?
\begin{enumerate}
\opcio{a}{$\{(x,y):x+y=1\}$}
\opcio{b}{$\{(x,y):xy=0\}$}
\opcio{c}{$\{(x,y):x^2+y^2=1\}$}
\opcio{d}{$\{(x,y):y=2x\}$}
\end{enumerate}\feedback
\pregunta El conjunt de polinomis reals de grau menor o igual que $4$ és:
\begin{enumerate}
\opcio{a}{Un conjunt sense estructura vectorial}
\opcio{b}{Un espai vectorial real}
\opcio{c}{Un espai vectorial només si s’exclou el polinomi nul}
\opcio{d}{Un cos}
\end{enumerate}\feedback
\pregunta Si $u,v\in W$ i $\lambda,\mu\in\mathbb K$, una manera compacta de provar que $W$ és subespai és comprovar que:
\begin{enumerate}
\opcio{a}{$u+v=0$}
\opcio{b}{$\lambda u=\mu v$}
\opcio{c}{$\lambda u+\mu v\in W$}
\opcio{d}{$u$ i $v$ són independents}
\end{enumerate}\feedback
\pregunta La intersecció de dos subespais d’un mateix espai vectorial és:
\begin{enumerate}
\opcio{a}{Sempre un subespai}
\opcio{b}{Mai un subespai}
\opcio{c}{Un subespai només si la intersecció és buida}
\opcio{d}{Un subespai només en dimensió dos}
\end{enumerate}\feedback
\pregunta Quin conjunt de matrius $2\times2$ reals és un subespai?
\begin{enumerate}
\opcio{a}{Les matrius amb determinant $1$}
\opcio{b}{Les matrius de traça $0$}
\opcio{c}{Les matrius invertibles}
\opcio{d}{Les matrius amb una entrada positiva}
\end{enumerate}\feedback
\pregunta El conjunt solució d’un sistema lineal homogeni és:
\begin{enumerate}
\opcio{a}{Sempre buit}
\opcio{b}{Una recta que no passa per l’origen}
\opcio{c}{Un conjunt que no conté el zero}
\opcio{d}{Un subespai vectorial}
\end{enumerate}\feedback
\pregunta Quina afirmació és falsa en general?
\begin{enumerate}
\opcio{a}{$V$ és subespai de si mateix}
\opcio{b}{$\{0\}$ és subespai de $V$}
\opcio{c}{La unió de dos subespais és sempre un subespai}
\opcio{d}{Tot subespai és tancat per producte per escalars}
\end{enumerate}\feedback
\end{enumerate}
